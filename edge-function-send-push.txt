// ── Supabase Edge Function: send-push ───────────────────────────
// 등록 위치: Supabase 대시보드 → Edge Functions → Create a new function
// 이름: send-push → 아래 전체 붙여넣기 → Deploy
// 이후: Database → Webhooks → Create webhook
//   table: reservations, events: INSERT → HTTP POST → 이 함수 URL
//
// Secrets 설정 (대시보드 → Edge Functions → Secrets):
//   VAPID_PUBLIC_KEY  = 공개키 값 (아래)
//   VAPID_PRIVATE_KEY = 비밀키 값 (아래)
//   SERVICE_ROLE_KEY  = 프로젝트 Settings → API → service_role 키

import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';
import webpush from 'npm:web-push@3';

const VAPID_PUBLIC = Deno.env.get('VAPID_PUBLIC_KEY')!;
const VAPID_PRIVATE = Deno.env.get('VAPID_PRIVATE_KEY')!;
const SERVICE_KEY = Deno.env.get('SERVICE_ROLE_KEY')!;
const BASE = Deno.env.get('SUPABASE_URL')!;
webpush.setVapidDetails('mailto:admin@example.com', VAPID_PUBLIC, VAPID_PRIVATE);

Deno.serve(async (req) => {
  try {
    const body = await req.json();
    const rec = body?.record ?? body;
    if (!rec || rec.user_id || !rec.owner) return new Response('skip', { status: 200 });

    const admin = createClient(BASE, SERVICE_KEY);
    const { data: subs } = await admin.from('push_subs').select('sub').eq('user_id', rec.owner);
    if (!subs?.length) return new Response('no subs', { status: 200 });

    const payload = JSON.stringify({
      title: '새 웹 접수 1건',
      body: `${rec.name ?? '손님'} · ${rec.date ?? ''} ${rec.time ?? ''}`.trim(),
      url: '/'
    });

    const stale: string[] = [];
    await Promise.allSettled(subs.map(async (s: any) => {
      try { await webpush.sendNotification(s.sub, payload); }
      catch (e: any) { if (e?.statusCode === 404 || e?.statusCode === 410) stale.push(s.sub.endpoint); }
    }));
    if (stale.length) await admin.from('push_subs').delete().in('endpoint', stale);
    return new Response('ok', { status: 200 });
  } catch (e) {
    return new Response('err', { status: 200 });
  }
});
